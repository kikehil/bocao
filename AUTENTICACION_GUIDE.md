```json
{
  "id": "customer_1738292485123",
  "name": "Juan Pérez",
  "email": "juan@test.com",
  "phone": "1234567890",
  "password": "test123",
  "createdAt": "2026-01-30T...",
  "savedAddresses": [],

  "orderHistory": []
```
